   145	/**
   146	 * Load `name` and compose its effective patch stack: bundle layers in
   147	 * `dsh.profile.bundles` order (a base-backed profile gets the base bundle's
   148	 * platform-gated shell rows), the profile's user layer, the home-level user
   149	 * layer (`$DSH_HOME/cordis.patch.yml` — machine-local preferences that apply
   150	 * to every profile, so it outranks the per-profile layer), `--patch` overlays,
   151	 * then the telemetry switch.
   152	 * @param name - the profile name.
   153	 * @param patchFiles - `--patch` overlay paths, in argv order.
   154	 * @returns the profile and its patch layers.
   155	 */
   156	async function composeProfile(
   157	  name: string,
   158	  patchFiles: readonly string[],
   159	): Promise<ComposedProfile> {
   160	  const profile = prepareProfile(name)
   161	  await healProfilesModuleFallback({ installAnchor: INSTALL_ANCHOR, profile })
   162	  const homePatches = loadOptionalPatches(NAME, homePatchPath()) ?? []
   163	  const overlays = patchFiles.flatMap(file => loadOverlayPatches(NAME, resolve(file)))
   164	  const bundlePatches = profile.layers.flatMap(layer => layer.patches)
   165	  const rows = new Map<string, EntryOptions>()
   166	  for (const row of composeEntries([bundlePatches, profile.patches, homePatches, overlays])) {
   167	    if (typeof row.id === 'string') rows.set(row.id, row)
   168	  }
   169	  const composedOverlays = [...overlays]
   170	  const telemetryPatch = resolveTelemetryPatch(process.env.DSH_TELEMETRY_DISABLED, rows.has(TELEMETRY_ROW_ID))
   171	  if (telemetryPatch !== undefined) composedOverlays.push(telemetryPatch)
   172	  return { profile, bundlePatches, homePatches, overlays: composedOverlays }

   805	export function loadProfile(
   806	  binName: string, name: string, installAnchor: string, home: string = resolveDshHome(),
   807	  options: { userLayer?: boolean } = {},
   808	): Profile {
   809	  const dir = resolveProfileDir(name, home)
   810	  if (!existsSync(join(dir, 'package.json'))) {
   811	    const template = PROFILE_TEMPLATES[name]
   812	    if (template === undefined) {
   813	      throw new Error(
   814	        `${binName}: profile ${JSON.stringify(name)} does not exist; create it with 'dsh plugin --profile ${name} add <package>'`,
   815	      )
   816	    }
   817	    initProfile(dir, template.bundles, template.patchReload)
   818	  }
   819	  const manifest = normalizeShippedProfile(name, dir, readProfileManifest(binName, dir))
   820	  // A hand-written profile manifest may omit the dsh section entirely.
   821	  const bundles = manifest.dsh?.profile?.bundles ?? []
   822	  const rawPatchReload: unknown = manifest.dsh?.profile?.patchReload
   823	  if (rawPatchReload !== undefined && rawPatchReload !== 'live' && rawPatchReload !== 'startup') {
   824	    throw new Error(
   825	      `${binName}: profile manifest ${join(dir, 'package.json')} dsh.profile.patchReload must be "live" or "startup"`,
   826	    )
   827	  }
   828	  const patchReload = rawPatchReload ?? DEFAULT_PROFILE_PATCH_RELOAD
   829	  const layers = bundles.map((packageName): ProfileLayer => {
   830	    const packageDir = resolveBundleDir(binName, packageName, installAnchor, dir)
   831	    const bundleManifest = JSON.parse(readFileSync(join(packageDir, 'package.json'), 'utf8')) as ProfileManifest
   832	    const declared = bundleManifest.dsh?.bundle?.patch
   833	    if (declared === undefined) {
   834	      throw new Error(`${binName}: profile bundle ${JSON.stringify(packageName)} declares no dsh.bundle in its package.json`)
   835	    }
   836	    const patchPath = join(packageDir, declared)
   837	    return { packageName, packageDir, patchPath, patches: loadOverlayPatches(binName, patchPath) }
   838	  })
   839	  const patchPath = join(dir, PROFILE_PATCH_FILENAME)
   840	  const patches = options.userLayer !== false && existsSync(patchPath)
   841	    ? loadOverlayPatches(binName, patchPath)

    65	 * command's action publishes the flags this invocation named; `--host 0.0.0.0`
    66	 * or a non-numeric `--port` is a usage error, so on rejection (and on `--help`)
    67	 * nothing is provided.
    68	 * @param ctx - plugin context carrying the command line.
    69	 */
    70	export function apply(ctx: Context): void {
    71	  const program = webCommand()
    72	  program.action(() => {
    73	    const options = program.opts<WebOptions>()
    74	    if (options.host === '0.0.0.0') {
    75	      program.error('error: --host 0.0.0.0 is intentionally not supported yet for safety: it would expose remote code execution to the network; use 127.0.0.1 instead')
    76	    }
    77	    if (options.port !== undefined && !/^\d+$/.test(options.port)) {
    78	      program.error(`error: --port must be a number, got ${JSON.stringify(options.port)}`)
    79	    }

    81	      openBrowser: options.open,
    82	      ...options.host !== undefined && { host: options.host },
    83	      ...options.port !== undefined && { port: Number(options.port) },
    84	      trustedHosts: options.trustedHost ?? [],
    85	    } satisfies WebStartupValues)
